def generated_tool(trajectory):
    left_tcp = trajectory.trace["left_tcp_position"]
    left_contact = trajectory.trace["roller_left_contact_position"]
    right_tcp = trajectory.trace["right_tcp_position"]
    right_contact = trajectory.trace["roller_right_contact_position"]
    physics_steps = trajectory.trace["physics_step"]

    terminal_index = len(left_tcp) - 1
    if terminal_index < 0:
        return {
            "value": None,
            "unit": "m",
            "passed": None,
            "evidence_steps": [],
            "details": {
                "operation": "derived_observable",
                "reason": "no_finite_sample",
            },
        }

    left_delta = left_tcp[terminal_index] - left_contact[terminal_index]
    right_delta = right_tcp[terminal_index] - right_contact[terminal_index]
    left_distance = np.linalg.norm(left_delta)
    right_distance = np.linalg.norm(right_delta)

    if not (np.isfinite(left_distance) and np.isfinite(right_distance)):
        return {
            "value": None,
            "unit": "m",
            "passed": None,
            "evidence_steps": [],
            "details": {
                "operation": "derived_observable",
                "reason": "no_finite_sample",
            },
        }

    value = left_distance if left_distance >= right_distance else right_distance
    return {
        "value": float(value),
        "unit": "m",
        "passed": None,
        "evidence_steps": [int(physics_steps[terminal_index])],
        "details": {
            "operation": "derived_observable",
            "reason": "measured",
        },
    }
